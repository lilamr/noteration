 72def _format_author(author: Any) -> str:
 73    if not author:
 74        return ""
 75    if isinstance(author, str):
 76        return author
 77    if isinstance(author, list):
 78        parts: list[str] = []
 79        for a in author:
 80            if isinstance(a, dict):
 81                family = a.get("family", "")
 82                given = a.get("given", "")
 83                combined = f"{family}, {given}".strip(", ")
 84                parts.append(combined)
 85            else:
 86                parts.append(str(a))
 87        return "; ".join(p for p in parts if p)
 88    return str(author)

 91def _parse_tags(raw_tags: Any) -> list[str]:

 99def _save_yaml(info_path: Path, data: dict[str, Any]) -> None:
100    yaml = get_yaml()
101    if not yaml:
102        raise RuntimeError("pyyaml is not installed; cannot save info.yaml")
103
104    tmp_path = info_path.with_suffix(".tmp")
105    try:
106        with open(tmp_path, "w", encoding="utf-8") as f:
107            yaml.dump(data, f, allow_unicode=True, default_flow_style=False, sort_keys=False)
108        tmp_path.replace(info_path)
109    except Exception as e:
110        logger.error(f"Failed to save YAML to {info_path}: {e}")
111        if tmp_path.exists():

116def _make_base_key(author: str, year: str, title: str) -> str:
117    first_author = author.split(";")[0].split(",")[0].strip()
118    first_author = re.sub(r"[^A-Za-z]", "", first_author)[:20]
119    yr = re.sub(r"[^0-9]", "", str(year))[:4]
120    
121    title_clean = re.sub(r"[^A-Za-z0-9\s]", "", title)
122    words = title_clean.split()
123    first_word = words[0][:15] if words else ""
124    
125    key = f"{first_author}{yr}{first_word}"
126    return key if key.strip() else "untitled"

149    def _load_with_cache(self, fts_engine: Any) -> list[LiteratureEntry]:
150        cached_data = fts_engine.get_all_literature_cache()
151        cache_map = {item["key"]: item for item in cached_data}
152
153        final_entries: list[LiteratureEntry] = []
154        found_keys: set[str] = set()
155
156        if self.library_path.exists():
157            for info_yaml in self.library_path.rglob("info.yaml"):
158                entry_key = info_yaml.parent.name
159                found_keys.add(entry_key)
160
161                try:
162                    mtime = info_yaml.stat().st_mtime
163                    cached = cache_map.get(entry_key)
164
165                    if cached and cached["mtime"] >= mtime:
166                        entry = self._entry_from_cache(cached)
167                        final_entries.append(entry)
168                    else:
169                        maybe_entry = self._parse_info_yaml(info_yaml)
170                        if maybe_entry:
171                            self._upsert_to_cache(fts_engine, maybe_entry, mtime)
172                            final_entries.append(maybe_entry)
173                except Exception as e:
174                    logger.error(f"Failed to sync literature entry {entry_key}: {e}")
175
176        stale_keys = set(cache_map.keys()) - found_keys
177        for key in stale_keys:
178            fts_engine.remove_literature_cache(key)
179
180        return final_entries

260    def search(self, query: str) -> list[LiteratureEntry]:
261        tokens = query.strip().split()
262        with self._lock:
263            results = self.all_entries()
264            for token in tokens:
265                if ":" in token:
266                    fname, _, value = token.partition(":")
267                    results = [
268                        e for e in results if self._match_field(e, fname.lower(), value.lower())
269                    ]
270                else:
271                    q = token.lower()
272                    results = [e for e in results if self._match_any(e, q)]
273            return results

299    def add_document(
300        self,
301        pdf_path: Path | None = None,
302        *,
303        title: str = "",
304        author: str = "",
305        year: str = "",
306        journal: str = "",
307        publisher: str = "",
308        doi: str = "",
309        isbn: str = "",
310        volume: str = "",
311        issue: str = "",
312        page: str = "",
313        abstract: str = "",
314        tags: list[str] | None = None,
315        collections: list[str] | None = None,
316        from_doi: str = "",
317        from_arxiv: str = "",
318        from_isbn: str = "",
319        fts_engine: Optional[Any] = None,
320        track_changes_callback: Optional[Callable[[Path], None]] = None,
321    ) -> LiteratureEntry | None:
322        with self._lock:
323            entry: LiteratureEntry | None = None
324
325            # mode 1: automation
326            has_automation = any([from_doi, from_arxiv, from_isbn])
327            if has_automation and self._papis_cli_available():
328                source = "doi" if from_doi else "arxiv" if from_arxiv else "isbn"
329                url = from_doi or from_arxiv or from_isbn
330                success, ref_key, is_duplicate = self._run_papis_add_from(source, url, pdf_path)
331                if success:
332                    self._entries = None
333                    if ref_key:
334                        entry = self.get(ref_key) or self.get_by_ref(ref_key)
335                    if not entry and not is_duplicate:
336                        entry = self._newest_entry()
337                    
338                    if entry:
339                        if is_duplicate:
340                            logger.info(f"Duplicate entry found: @{entry.key}, skipping rename.")
341                            return entry
342                        
343                        # Apply metadata overrides
344                        for field, val in [
345                            ("title", title),
346                            ("author", author),
347                            ("year", year),
348                            ("journal", journal),
349                            ("publisher", publisher),
350                            ("doi", doi),
351                        ]:
352                            if val:
353                                self.update_field(
354                                    entry.key,
355                                    field,
356                                    val,
357                                    fts_engine=fts_engine,
358                                    track_changes_callback=track_changes_callback,
359                                )
360                        
361                        # Re-fetch entry to get updated metadata for naming
362                        entry = self.get(entry.key)
363                        if entry:
364                            # Rename to standard format if it's currently a hash/uuid
365                            # Using 'papis rename' is the standard way to handle this
366                            if len(entry.key) > 20: # Heuristic for hash
367                                # 1. Generate desired key and set it as 'ref' to help Papis
368                                desired_key = _make_base_key(entry.author, entry.year, entry.title)
369                                self.update_field(entry.key, "ref", desired_key)
370                                
371                                logger.info(f"Normalizing folder name for entry @{entry.key}...")
372                                try:
373                                    # 2. Use papis rename with --doc-folder to target it directly
374                                    # We use {doc[ref]} template which Papis 0.15+ supports reliably
375                                    if self._run_papis_rename(entry.key, "{doc[ref]}"):
376                                        # After papis rename, the folder name changes. 
377                                        # We need to find the new key.
378                                        self._entries = None
379                                        new_entry = self._newest_entry()
380                                        if new_entry:
381                                            entry = new_entry
382                                            logger.info(f"Folder successfully renamed to @{entry.key}")
383                                except Exception as e:
384                                    logger.error(f"Failed to rename imported folder {entry.key} via papis: {e}")
385                        
386                        return entry
387
388            # mode 2: manual add
389            key = self._unique_key(_make_base_key(author, year, title))
390            entry_dir = self.library_path / key
391            entry_dir.mkdir(parents=True, exist_ok=True)
392            data: dict[str, Any] = {
393                "ref": key,
394                "title": title,
395                "author": author,
396                "year": year,
397                "journal": journal,
398                "publisher": publisher,
399                "doi": doi,
400                "isbn": isbn,
401                "volume": volume,
402                "issue": issue,
403                "page": page,
404                "abstract": abstract,
405            }
406            if tags:
407                data["tags"] = tags
408            if collections:
409                data["collections"] = collections
410            info_path = entry_dir / "info.yaml"
411            _save_yaml(info_path, data)
412
413            pdf_dest: Path | None = None
414            if pdf_path and pdf_path.exists():
415                pdf_dest = entry_dir / f"{key}.pdf"
416                shutil.copy2(pdf_path, pdf_dest)
417                data["files"] = [f"{key}.pdf"]
418                _save_yaml(info_path, data)
419
420            entry = LiteratureEntry(
421                key=key,
422                title=title,
423                author=author,
424                year=year,
425                journal=journal,
426                publisher=publisher,
427                doi=doi,
428                isbn=isbn,
429                volume=volume,
430                issue=issue,
431                page=page,
432                abstract=abstract,
433                tags=tags or [],
434                collections=collections or [],
435                pdf_path=pdf_dest,
436                info_path=info_path,
437                _raw=data,
438            )
439            if self._entries is not None:
440                self._entries.append(entry)
441            if fts_engine:
442                self._upsert_to_cache(fts_engine, entry, info_path.stat().st_mtime)
443                fts_engine.index_tags(key, tags or [], source="literature")
444            if track_changes_callback:
445                track_changes_callback(info_path)
446            return entry

628    def _run_papis_add_from(
629        self, source: str, url: str, pdf_path: Path | None = None
630    ) -> tuple[bool, str, bool]:
631        try:
632            papis_bin = shutil.which("papis")
633            cmd = [papis_bin] if papis_bin else [sys.executable, "-m", "papis"]
634            cmd += [
635                "--lib",
636                str(self.library_path),
637                "add",
638                "--from",
639                source,
640                url,
641                "--batch",
642                "--no-download-files",
643                "--folder-name",
644                "{author}{year}{title}",
645            ]
646            if pdf_path and pdf_path.exists():
647                cmd.append(str(pdf_path))
648            result = subprocess.run(cmd, capture_output=True, timeout=60, encoding="utf-8")
649            out = (result.stdout or "") + (result.stderr or "")
650            is_dup = "already in your library" in out or "Duplication Warning" in out
651            ref = re.search(r"ref:\s*(\S+)", out)
652            return result.returncode == 0, ref.group(1) if ref else "", is_dup
653        except Exception:
654            return False, "", False
