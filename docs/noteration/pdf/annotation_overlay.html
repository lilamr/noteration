noteration.pdf.annotation_overlay

noteration/pdf/annotation_overlay.py

Transparent QWidget placed on top of a PDF page to display:

  • Highlights (semi-transparent colored rects)
  • Comments (sticky note icon)
  • Bookmarks (bookmark icon)

Interaction:

  • Right-click on overlay → context menu (delete, edit, jump to note)
  • Drag to create a new highlight (when highlight mode is active)
  • Click to add a comment (when comment mode is active)

122    def _extract_text_from_rect(self, rect_pts: list[float]) -> str:
123        """Extract text from a rectangle using cached words.
124        Fallback to fitz if words are missing (not recommended for main thread).
125        """
126        if not self._page_words:
127            # If we don't have words yet, we can't do much without blocking.
128            # In a proper async flow, we should already have them.
129            return ""
130            
131        x0, y0, x1, y1 = rect_pts
132        text_parts = []
133        for w in self._page_words:
134            # words: (x0, y0, x1, y1, "word", ...)
135            if w[0] >= x0 and w[1] >= y0 and w[2] <= x1 and w[3] <= y1:
136                text_parts.append(w[4])
137        
138        return " ".join(text_parts)

162    def _scale(self) -> float:
163        """Scale factor: pixel / PDF point."""
164        if self._page_w <= 0 or self.width() <= 0:
165            return 1.0
166        return self.width() / self._page_w
167
168    def _pts_to_px(self, bbox: list[float]) -> QRectF:
169        """Convert coordinates from PDF points to pixel coordinates."""
170        s = self._scale()
171        x0, y0, x1, y1 = bbox
172        return QRectF(x0 * s, y0 * s, (x1 - x0) * s, (y1 - y0) * s)
173
174    def _px_to_pts(self, px_rect: QRectF) -> list[float]:
175        """Convert coordinates from pixel coordinates to PDF points."""
176        s = self._scale()
177        if s <= 0:
178            s = 1.0
179        return [
180            px_rect.x() / s,
181            px_rect.y() / s,
182            px_rect.right() / s,
183            px_rect.bottom() / s,
184        ]

190    def _find_word_at_pos(self, pos_pts: QPointF) -> int:
191        """Find nearest word index to position (in PDF points)."""
192        if not self._page_words:
193            return -1
194
195        px, py = pos_pts.x(), pos_pts.y()
196        best_idx = -1
197        min_dist = 1e9
198
199        # 1. Priority: Is the point inside a word box?
200        for i, w in enumerate(self._page_words):
201            # w: (x0, y0, x1, y1, ...)
202            if w[0] <= px <= w[2] and w[1] <= py <= w[3]:
203                return i
204
205        # 2. Secondary: Search for nearest word horizontally with slight vertical tolerance.
206        # Helps when highlighting between lines or at word edges.
207        for i, w in enumerate(self._page_words):
208            cx, cy = (w[0] + w[2]) / 2, (w[1] + w[3]) / 2
209            # Apply heavy penalty to vertical distance (dy) to prevent line jumping.
210            dx = abs(px - cx)
211            dy = abs(py - cy)
212
213            dist = dx + dy * 4.0
214            if dist < min_dist:
215                min_dist = dist
216                best_idx = i
217
218        # Tolerance increased to 150 points (~2 inches) to ensure "snap" works.
219        return best_idx if min_dist < 150 else -1

221    def paintEvent(self, event: QPaintEvent) -> None:
222        """Paint the annotation overlay."""
223        painter = QPainter(self)
224        try:
225            painter.setRenderHint(QPainter.RenderHint.Antialiasing)
226
227            doc = self._store.load(self.papis_key)
228            for ann in doc.for_page(self.page_idx):
229                self._paint_annotation(painter, ann)
230
231            # Selection preview
232            if self._dragging and self._drag_start_pos and self._drag_end_pos:
233                if self._mode == "highlight":
234                    if self._drag_start_idx != -1 and self._drag_end_idx != -1:
235                        self._paint_selection_preview(painter)
236                    else:
237                        # Fallback rect preview
238                        drag_rect = QRectF(self._drag_start_pos, self._drag_end_pos).normalized()
239                        painter.setBrush(QBrush(QColor(255, 235, 59, 80)))
240                        painter.setPen(QPen(QColor(255, 193, 7), 1))
241                        painter.drawRect(drag_rect)
242                elif self._mode == "image":
243                    drag_rect = QRectF(self._drag_start_pos, self._drag_end_pos).normalized()
244                    painter.setBrush(QBrush(QColor(255, 235, 59, 80)))
245                    painter.setPen(QPen(QColor(255, 193, 7), 1))
246                    painter.drawRect(drag_rect)
247        finally:
248            painter.end()

250    def _paint_selection_preview(self, painter: QPainter) -> None:
251        """Draw text selection preview that follows lines precisely."""
252        start = min(self._drag_start_idx, self._drag_end_idx)
253        end = max(self._drag_start_idx, self._drag_end_idx)
254
255        s = self._scale()
256        color = QColor(255, 235, 59, 130)
257        painter.setBrush(QBrush(color))
258        painter.setPen(Qt.PenStyle.NoPen)
259
260        # Group selected words into lines for clean rendering
261        lines: dict[int, list[tuple]] = {}
262        for i in range(start, end + 1):
263            w_data = self._page_words[i]
264            l_key = w_data[6]  # line_no
265            if l_key not in lines:
266                lines[l_key] = []
267            lines[l_key].append(w_data)
268
269        for l_no in sorted(lines.keys()):
270            l_words = lines[l_no]
271            lx0 = min(item[0] for item in l_words)
272            ly0 = min(item[1] for item in l_words)
273            lx1 = max(item[2] for item in l_words)
274            ly1 = max(item[3] for item in l_words)
275            painter.drawRect(QRectF(lx0 * s, ly0 * s, (lx1 - lx0) * s, (ly1 - ly0) * s))

277    def _paint_annotation(self, painter: QPainter, ann: Annotation) -> None:
278        """Paint a single annotation based on its type."""
279        if ann.type in ("highlight", "image"):
280            color = QColor(ann.color)
281            color.setAlpha(90)
282            painter.setBrush(QBrush(color))
283            border_color = QColor(ann.color).darker(140)
284            border_color.setAlpha(160)
285            painter.setPen(QPen(border_color, 0.5))
286
287            # Prioritize quads for highlight (text)
288            if ann.type == "highlight" and ann.quads:
289                s = self._scale()
290                path = QPainterPath()
291                for q in ann.quads:
292                    # q: [p0.x, p0.y, p1.x, p1.y, p2.x, p2.y, p3.x, p3.y]
293                    # Order p0(TL), p1(TR), p2(BL), p3(BR)
294                    # Polygon must: TL -> TR -> BR -> BL
295                    poly = QPolygonF(
296                        [
297                            QPointF(q[0] * s, q[1] * s),  # p0
298                            QPointF(q[2] * s, q[3] * s),  # p1
299                            QPointF(q[6] * s, q[7] * s),  # p3
300                            QPointF(q[4] * s, q[5] * s),  # p2
301                        ]
302                    )
303                    path.addPolygon(poly)
304                painter.drawPath(path)
305            # Fallback to rect for image or if highlight quads missing
306            elif ann.rect:
307                rect = self._pts_to_px(ann.rect)
308                painter.drawRect(rect)
309
310            # Small tag label if a note exists
311            if ann.note and ann.rect:
312                rect = self._pts_to_px(ann.rect)
313                painter.setFont(QFont("Arial", 8))
314                painter.setPen(QPen(QColor("#555"), 1))
315                painter.drawText(int(rect.x()), int(rect.y() - 2), ann.note[:30])
316
317        elif ann.type == "comment" and ann.position:
318            s = self._scale()
319            cx = ann.position[0] * s
320            cy = ann.position[1] * s
321            # Sticky note icon (18×18)
322            icon_rect = QRectF(cx - 9, cy - 9, 18, 18)
323            painter.setBrush(QBrush(QColor("#FFF9C4")))
324            painter.setPen(QPen(QColor("#F9A825"), 1.5))
325            painter.drawRoundedRect(icon_rect, 2, 2)
326            painter.setFont(QFont("Arial", 7))
327            painter.setPen(QPen(QColor("#555")))
328            painter.drawText(icon_rect, Qt.AlignmentFlag.AlignCenter, "✏")
329
330        elif ann.type == "bookmark":
331            if ann.position:
332                s = self._scale()
333                cx = ann.position[0] * s
334                cy = 0
335                painter.setBrush(QBrush(QColor("#EF9F27")))
336                painter.setPen(Qt.PenStyle.NoPen)
337                # Bookmark triangle
338                triangle = QPolygonF(
339                    [
340                        QPointF(cx - 6, cy),
341                        QPointF(cx + 6, cy),
342                        QPointF(cx, cy + 12),
343                    ]
344                )
345                painter.drawPolygon(triangle)

351    def mousePressEvent(self, event: QMouseEvent) -> None:
352        """Handle mouse press events for annotation interactions."""
353        pos = event.position()
354        s = self._scale()
355        pos_pts = QPointF(pos.x() / s, pos.y() / s)
356
357        if event.button() == Qt.MouseButton.LeftButton:
358            if self._mode in ("highlight", "image"):
359                self._drag_start_pos = pos
360                self._drag_end_pos = pos
361                self._dragging = True
362
363                if self._mode == "highlight":
364                    self._drag_start_idx = self._find_word_at_pos(pos_pts)
365                    self._drag_end_idx = self._drag_start_idx
366
367            elif self._mode == "comment":
368                self._add_comment(pos_pts.x(), pos_pts.y())
369
370            elif self._mode == "view":
371                # Check click on existing annotation
372                ann = self._hit_test(pos)
373                if ann:
374                    self._show_annotation_detail(ann)
375
376        elif event.button() == Qt.MouseButton.RightButton:
377            ann = self._hit_test(pos)
378            self._show_context_menu(event.globalPosition().toPoint(), ann)

380    def mouseMoveEvent(self, event: QMouseEvent) -> None:
381        """Handle mouse move events for dragging/selection."""
382        if self._dragging:
383            self._drag_end_pos = event.position()
384
385            if self._mode == "highlight":
386                s = self._scale()
387                pos_pts = QPointF(event.position().x() / s, event.position().y() / s)
388                self._drag_end_idx = self._find_word_at_pos(pos_pts)

392    def mouseReleaseEvent(self, event: QMouseEvent) -> None:
393        """Handle mouse release events to finalize selection or action."""
394        if self._dragging:
395            self._dragging = False
396
397            if self._mode == "highlight":
398                if self._drag_start_idx != -1 and self._drag_end_idx != -1:
399                    self._finalize_text_selection()
400                elif self._drag_start_pos and self._drag_end_pos:
401                    # FALLBACK: If no word detected, use box selection
402                    drag_rect = QRectF(self._drag_start_pos, self._drag_end_pos).normalized()
403                    if drag_rect.width() > 5 and drag_rect.height() > 5:
404                        pts = self._px_to_pts(drag_rect)
405                        self._add_highlight(pts)
406            elif self._mode == "image":
407                if self._drag_start_pos and self._drag_end_pos:
408                    drag_rect = QRectF(self._drag_start_pos, self._drag_end_pos).normalized()
409                    if drag_rect.width() > 5 and drag_rect.height() > 5:
410                        pts = self._px_to_pts(drag_rect)
411                        self._add_image_highlight(pts)
412
413            self._drag_start_pos = None
414            self._drag_end_pos = None
415            self._drag_start_idx = -1
416            self._drag_end_idx = -1
417            self.update()

419    def _finalize_text_selection(self) -> None:
420        """Collect quads and text from selected words and save."""
421        start = min(self._drag_start_idx, self._drag_end_idx)
422        end = max(self._drag_start_idx, self._drag_end_idx)
423
424        words = self._page_words[start : end + 1]
425        if not words:
426            return
427
428        quads = []
429        text_parts = []
430        lines: dict[int, list[tuple]] = {}
431
432        # Global bounding box for rect
433        all_x0, all_y0 = 1e9, 1e9
434        all_x1, all_y1 = -1e9, -1e9
435
436        for w in words:
437            # (x0, y0, x1, y1, text, b_no, l_no, w_no)
438            all_x0 = min(all_x0, w[0])
439            all_y0 = min(all_y0, w[1])
440            all_x1 = max(all_x1, w[2])
441            all_y1 = max(all_y1, w[3])
442
443            l_key = w[6]
444            if l_key not in lines:
445                lines[l_key] = []
446            lines[l_key].append(w)
447            text_parts.append(w[4])
448
449        for l_no in sorted(lines.keys()):
450            l_words = lines[l_no]
451            lx0 = min(w[0] for w in l_words)
452            ly0 = min(w[1] for w in l_words)
453            lx1 = max(w[2] for w in l_words)
454            ly1 = max(w[3] for w in l_words)
455            # p0, p1, p2, p3 (TL, TR, BL, BR)
456            quads.append([lx0, ly0, lx1, ly0, lx0, ly1, lx1, ly1])
457
458        text = " ".join(text_parts)
459        rect_pts = [all_x0, all_y0, all_x1, all_y1]
460
461        ann = self._store.new_highlight(
462            papis_key=self.papis_key,
463            page=self.page_idx,
464            rect=rect_pts,
465            quads=quads,
466            text_content=text,
467        )
468        self.annotation_created.emit(ann)

474    def _add_highlight(self, rect_pts: list[float]) -> None:
475        """Create and save a new highlight annotation."""
476        try:
477            x0, y0, x1, y1 = rect_pts
478            quads: list[list[float]] = []
479            text_parts: list[str] = []
480
481            # Get all words within the rect area (with 1pt tolerance)
482            # words: (x0, y0, x1, y1, "word", block_no, line_no, word_no)
483            words_in_rect = [
484                w for w in self._page_words
485                if w[0] >= x0 - 1 and w[1] >= y0 - 1 and w[2] <= x1 + 1 and w[3] <= y1 + 1
486            ]
487
488            # Group by line_no to create quads per line
489            lines: dict[int, list[Any]] = {}
490            for w in words_in_rect:
491                l_no = w[6]
492                if l_no not in lines:
493                    lines[l_no] = []
494                lines[l_no].append(w)
495                text_parts.append(w[4])
496
497            for l_no in sorted(lines.keys()):
498                line_words = lines[l_no]
499                # Combine words in one line into a single bounding box
500                lx0 = min(w[0] for w in line_words)
501                ly0 = min(w[1] for w in line_words)
502                lx1 = max(w[2] for w in line_words)
503                ly1 = max(w[3] for w in line_words)
504
505                # Store as p0, p1, p2, p3
506                quads.append([lx0, ly0, lx1, ly0, lx0, ly1, lx1, ly1])
507
508            text = " ".join(text_parts)
509
510            ann = self._store.new_highlight(
511                papis_key=self.papis_key,
512                page=self.page_idx,
513                rect=rect_pts,
514                quads=quads if quads else None,
515                text_content=text or "",
516            )
517            self.update()
518            self.annotation_created.emit(ann)
519        except Exception as e:
520            logger.error(f"Failed to create highlight: {e}")

564    def _hit_test(self, pos: QPointF) -> Annotation | None:
565        """Perform hit testing to identify an annotation under the cursor."""
566        doc = self._store.load(self.papis_key)
567        for ann in reversed(doc.for_page(self.page_idx)):
568            if ann.type in ("highlight", "image") and ann.rect:
569                rect = self._pts_to_px(ann.rect)
570                if rect.contains(pos):
571                    return ann
572            elif ann.type == "comment" and ann.position:
573                s = self._scale()
574                cx = ann.position[0] * s
575                cy = ann.position[1] * s
576                if abs(pos.x() - cx) < 12 and abs(pos.y() - cy) < 12:
577                    return ann
578        return None
