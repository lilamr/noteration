 43class SyncWorker(QObject):
 44    log_line = Signal(str, str)  # (text, level)
 45    finished = Signal(object)  # SyncResult
 46
 47    def __init__(self, vault: "VaultManager") -> None:
 48        super().__init__()
 49        self._vault = vault
 50        self._vault_path = vault.vault_path
 51        self._storage_path = vault.storage_path
 52        self._repo = vault.core.git_repo
 53        self._config = vault.config
 54        self._op = "sync"
 55
 56    def set_operation(self, op: str) -> None:
 57        self._op = op
 58
 59    def run(self) -> None:
 60        def log(msg: str) -> None:
 61            lvl = (
 62                "ok"
 63                if msg.startswith("  ✓") or msg.startswith("✓")
 64                else "error"
 65                if "✗" in msg or "ERROR" in msg
 66                else "warn"
 67                if "⚠" in msg
 68                else "info"
 69            )
 70            self.log_line.emit(msg, lvl)
 71
 72        # Handle Encrypted Vault Sync Lifecycle
 73        is_encrypted = self._config.get("security", "encryption_enabled", False)
 74        if is_encrypted and self._op == "sync":
 75            log("🛡️ Encrypted vault detected. Preparing for sync...")
 76            if not self._vault.secret_key:
 77                log(
 78                    "✗ Error: No private key available in this session. Cannot decrypt incoming changes."
 79                )
 80                self.finished.emit(
 81                    SyncResult(status=SyncStatus.ERROR, message="Missing private key")
 82                )
 83                return
 84
 85            # 1. PRE-SYNC ENCRYPTION
 86            try:
 87                log("  → Encrypting all vault contents...")
 88                self._vault.core.encrypt_vault(log_callback=log)
 89                log("  ✓ All changes encrypted (Staging)")
 90            except Exception as e:
 91                log(f"✗ Pre-sync encryption failed: {e}")
 92                self.finished.emit(SyncResult(status=SyncStatus.ERROR, message=str(e)))
 93                return
 94
 95        if self._op == "init":
 96            try:
 97                GitRepo.init(self._storage_path)
 98                log("✓ Git repository initialized")
 99                self.finished.emit(SyncResult(status=SyncStatus.SUCCESS, message="Init complete"))
100            except Exception as e:
101                log(f"✗ {e}")
102                self.finished.emit(SyncResult(status=SyncStatus.ERROR, message=str(e)))
103            return
104
105        if self._op == "abort":
106            if self._repo and self._repo.abort_sync():
107                log("✓ Synchronization aborted")
108                self.finished.emit(SyncResult(status=SyncStatus.SUCCESS, message="Aborted"))
109            else:
110                log("✗ Failed to abort synchronization")
111                self.finished.emit(SyncResult(status=SyncStatus.ERROR, message="Abort failed"))
112            return
113
114        if self._op == "continue":
115            if self._repo:
116                result = self._repo.continue_sync(log_callback=log)
117                self.finished.emit(result)
118            return
119
120        # Normal sync
121        remote = self._config.get("sync", "remote", "origin")
122        branch = self._config.get("sync", "branch", "")
123
124        strat_s = self._config.get("sync", "strategy", "rebase")
125        strategy = {"merge": SyncStrategy.MERGE, "stash": SyncStrategy.STASH}.get(
126            strat_s, SyncStrategy.REBASE
127        )
128
129        if self._repo:
130            try:
131                # Ensure .gitignore is up to date and metadata is untracked before sync
132                self._repo.ensure_ignored()
133
134                result = self._repo.sync(
135                    remote=remote,
136                    branch=branch,
137                    strategy=strategy,
138                    log_callback=log,
139                )
140
141                # 2. POST-SYNC DECRYPTION (if sync was successful)
142                if is_encrypted and result.status == SyncStatus.SUCCESS:
143                    try:
144                        log("🛡️ Sync complete. Decrypting new incoming changes...")
145                        self._vault.core.decrypt_vault(log_callback=log)
146                        log("  ✓ Incoming changes decrypted and available")
147                    except Exception as e:
148                        log(f"⚠ Warning: Post-sync decryption failed: {e}")
149
150                self.finished.emit(result)
151            except Exception as e:
152                log(f"✗ Unexpected error: {e}")
153                self.finished.emit(SyncResult(status=SyncStatus.ERROR, message=str(e)))

190class LogEdit(QPlainTextEdit):
191    def __init__(self, parent=None):
192        super().__init__(parent)
193        self.setReadOnly(True)

197    def append_log(self, text: str, level: str = "info"):
198        fmt = QTextCharFormat()
199        if level == "error":
200            fmt.setForeground(QColor("#ff5f57"))
201        elif level == "ok":
202            fmt.setForeground(QColor("#28c840"))
203        elif level == "warn":
204            fmt.setForeground(QColor("#febc2e"))
205        else:
206            fmt.setForeground(QColor("#a0a0a8"))
207
208        self.setCurrentCharFormat(fmt)
209        self.appendPlainText(text)
210        self.moveCursor(QTextCursor.MoveOperation.End)

218        super().__init__(parent)

275    def _refresh_status(self):
276        repo = self.vault.core.git_repo
277        if not repo:
278            self._lbl_branch.setText("Not a Git repository")
279            self._btn_sync.setText("Initialize Git")
280            return
281
282        status = repo.status(session_hashes=self.vault.core.session_hashes)
283        self._lbl_branch.setText(status.branch)
284        self._lbl_remote.setText(", ".join(status.remotes) if status.remotes else "None")
285        self._lbl_status.setText("Dirty (Uncommitted)" if status.is_dirty else "Clean")
286        self._lbl_ahead.setText(str(status.ahead))
287        self._lbl_behind.setText(str(status.behind))
288        self._lbl_last.setText(status.last_commit_time or "Never")
289        self._btn_sync.setText("Sync Now")
290
291    def start_sync(self):
292        if not self.vault.core.git_repo:
293            self._start_op("init", "Initializing...")
294            return
295        self._start_op("sync", "Synchronizing...")
296
297    def _on_sync(self):
298        self.start_sync()
299
300    def _start_op(self, op: str, status_text: str):
301        if self._thread and self._thread.isRunning():
302            return
303
304        self._log_edit.append_log(f">>> Starting {op.upper()}...")
305        self._btn_sync.setEnabled(False)
306        self._btn_sync.setText(f"{status_text}...")
307        self._btn_sync.setStyleSheet("font-weight: bold; background: #6c757d; color: white;")
308
309        self._thread = QThread()
310        self._worker = SyncWorker(self.vault)
311        self._worker.set_operation(op)
312        self._worker.moveToThread(self._thread)
313
314        self._worker.log_line.connect(self._log_edit.append_log)
315        self._worker.finished.connect(self._on_finished)
316        self._thread.started.connect(self._worker.run)
317
318        self._thread.start()
319
320    def _on_finished(self, result: SyncResult):
321        if self._thread:
322            self._thread.quit()
323            self._thread.wait()
324
325        # 1. Handle Conflicts
326        if result.status == SyncStatus.CONFLICT and result.conflicts:
327            from noteration.dialogs.conflict_dialog import ConflictResolutionDialog
328
329            dlg = ConflictResolutionDialog(result.conflicts, self)
330            if dlg.exec():
331                resolutions = dlg.get_resolutions()
332                repo = self.vault.core.git_repo
333                if repo:
334                    self._log_edit.append_log("Applying resolutions...", "info")
335                    public_key = self.vault.config.get("security", "public_key", "")
336                    for path, content in resolutions.items():
337                        repo.resolve_conflict(path, content, public_key=public_key)
338
339                    # Continue the sync process (commit/push)
340                    self._start_op("continue", "Completing sync")
341                    return
342            else:
343                self._log_edit.append_log("⚠ Sync aborted. Conflicts remain unresolved.", "warn")
344
345        # 2. Post-op cleanup and status update
346        if result.ok:
347            self.vault.core.refresh_git_repo()
348
349        self._btn_sync.setEnabled(True)
350        self._btn_sync.setText("Sync Now")
351        self._btn_sync.setStyleSheet("font-weight: bold; background: #2d6a4f; color: white;")
352        self._refresh_status()
353        self._log_edit.append_log(f"--- {result.message} ---", "ok" if result.ok else "error")
354
355        # Trigger global status update so MainWindow badge updates
356        self.vault.request_git_status()
357
358    def shutdown(self) -> None:
359        """Safely stop any background sync threads."""
360        if self._thread and self._thread.isRunning():
361            logger.info("SyncTab: Stopping background sync thread...")
362            self._thread.requestInterruption()
363            self._thread.quit()
364            if not self._thread.wait(5000):
365                logger.warning("SyncTab: Sync thread failed to stop within timeout.")
366        
367        # We don't set to None immediately if we just called wait to be extra safe
368        # but in shutdown it is generally okay as the object is being destroyed.
369        self._thread = None
370        self._worker = None
371
372    def _on_set_remote(self):
373        repo = self.vault.core.git_repo
374
375        # If no repo exists, current remotes is empty
376        current_remotes = repo.list_remotes() if repo else []
377
378        dlg = SetRemoteDialog(current_remotes, self)
379        if dlg.exec():
380            name, url = dlg.result_remote()
381
382            if not name or not url:
383                self._log_edit.append_log("✗ Error: Remote name and URL cannot be empty.", "error")
384                return
385
386            if repo:
387                try:
388                    repo.add_remote(name, url)
389                    self._log_edit.append_log(f"✓ Remote '{name}' set to: {url}", "ok")
390                    self._refresh_status()
391                except Exception as e:
392                    self._log_edit.append_log(f"✗ Failed to set remote: {e}", "error")
393            else:
394                # If no repo exists, initialize it now with this remote
395                try:
396                    from noteration.sync.git_engine import GitRepo
397
398                    GitRepo.init(self.vault.storage_path, remote_url=url)
399                    self.vault.core.refresh_git_repo()
400                    self._log_edit.append_log(
401                        f"✓ Git repository initialized with remote: {url}", "ok"
402                    )
403                    self._refresh_status()
404                except Exception as e:
405                    self._log_edit.append_log(f"✗ Failed to initialize Git: {e}", "error")

321    def _on_finished(self, result: SyncResult):

330            dlg = ConflictResolutionDialog(result.conflicts, self)
331            if dlg.exec():
332                resolutions = dlg.get_resolutions()
333                repo = self.vault.core.git_repo
334                if repo:
335                    self._log_edit.append_log("Applying resolutions...", "info")
336                    public_key = self.vault.config.get("security", "public_key", "")
337                    for path, content in resolutions.items():

339
340                    # Continue the sync process (commit/push)
341                    self._start_op("continue", "Completing sync")
342                    return

345
346        # 2. Post-op cleanup and status update
347        if result.ok:
348            self.vault.core.refresh_git_repo()
349
350        self._btn_sync.setEnabled(True)

352        self._btn_sync.setStyleSheet("font-weight: bold; background: #2d6a4f; color: white;")
353        self._refresh_status()

357        self.vault.request_git_status()
