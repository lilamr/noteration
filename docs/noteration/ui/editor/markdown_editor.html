141    def line_number_area_paint_event(self, event) -> None:
142        painter = QPainter(self._lnum_area)
143        painter.fillRect(
144            event.rect(),
145            self.palette().color(QPalette.ColorRole.Window),
146        )
147        block = self.firstVisibleBlock()
148        block_num = block.blockNumber()
149        top = round(self.blockBoundingGeometry(block).translated(self.contentOffset()).top())
150        bottom = top + round(self.blockBoundingRect(block).height())
151        current = self.textCursor().blockNumber()
152
153        while block.isValid() and top <= event.rect().bottom():
154            if block.isVisible() and bottom >= event.rect().top():
155                color = QColor("#444") if block_num == current else QColor("#bbb")
156                painter.setPen(color)
157                painter.drawText(
158                    0,
159                    top,
160                    self._lnum_area.width() - 4,
161                    self.fontMetrics().height(),
162                    Qt.AlignmentFlag.AlignRight,
163                    str(block_num + 1),
164                )
165            block = block.next()
166            top = bottom
167            bottom = top + round(self.blockBoundingRect(block).height())
168            block_num += 1

202    def keyPressEvent(self, event) -> None:
203        if not self._vim_enabled:
204            if event.key() == Qt.Key.Key_Tab:
205                self.insertPlainText(" " * self.config.get("editor", "tab_width", 2))
206                return
207            if event.modifiers() == Qt.KeyboardModifier.ControlModifier:
208                if event.key() == Qt.Key.Key_V:
209                    clipboard = QApplication.clipboard()
210                    image = clipboard.image()
211                    if not image.isNull():
212                        self.image_pasted.emit(image)
213                        return
214                    mime = clipboard.mimeData()
215                    if mime.hasUrls():
216                        for url in mime.urls():
217                            if url.isLocalFile():
218                                path = url.toLocalFile()
219                                ext = path.lower().rsplit(".", 1)[-1] if "." in path else ""
220                                if ext in ("png", "jpg", "jpeg", "gif", "webp", "bmp"):
221                                    self.image_dropped.emit(path)
222                                    return
223                elif event.key() == Qt.Key.Key_Y:
224                    self.redo()
225                    return
226            super().keyPressEvent(event)
227            return
228
229        # Vim handling
230        key = event.key()
231        text = event.text()
232
233        if key == Qt.Key.Key_Escape:
234            if self._vim_mode == VimMode.NORMAL:
235                self.vim_exit_requested.emit()
236
237            self._set_vim_mode(VimMode.NORMAL)
238            cursor = self.textCursor()
239            cursor.clearSelection()
240            self.setTextCursor(cursor)
241            return
242
243        if self._vim_mode == VimMode.INSERT:
244            super().keyPressEvent(event)
245            return
246
247        # NORMAL / VISUAL / LINE_VISUAL handling
248        cursor = self.textCursor()
249        move_mode = QTextCursor.MoveMode.MoveAnchor
250        if self._vim_mode in (VimMode.VISUAL, VimMode.LINE_VISUAL):
251            move_mode = QTextCursor.MoveMode.KeepAnchor
252
253        # Allow standard shortcuts (Ctrl+S, Ctrl+C, etc.) even in Normal mode
254        if event.modifiers() == Qt.KeyboardModifier.ControlModifier:
255            super().keyPressEvent(event)
256            return
257
258        # Movement keys
259        if key == Qt.Key.Key_H:
260            cursor.movePosition(QTextCursor.MoveOperation.Left, move_mode)
261        elif key == Qt.Key.Key_L:
262            cursor.movePosition(QTextCursor.MoveOperation.Right, move_mode)
263        elif key == Qt.Key.Key_J:
264            cursor.movePosition(QTextCursor.MoveOperation.Down, move_mode)
265        elif key == Qt.Key.Key_K:
266            cursor.movePosition(QTextCursor.MoveOperation.Up, move_mode)
267        elif key == Qt.Key.Key_W:
268            cursor.movePosition(QTextCursor.MoveOperation.NextWord, move_mode)
269        elif key == Qt.Key.Key_B:
270            cursor.movePosition(QTextCursor.MoveOperation.PreviousWord, move_mode)
271        elif key == Qt.Key.Key_0:
272            cursor.movePosition(QTextCursor.MoveOperation.StartOfLine, move_mode)
273        elif key == Qt.Key.Key_Dollar:
274            cursor.movePosition(QTextCursor.MoveOperation.EndOfLine, move_mode)
275        elif key == Qt.Key.Key_G:
276            if event.modifiers() & Qt.KeyboardModifier.ShiftModifier:
277                cursor.movePosition(QTextCursor.MoveOperation.End, move_mode)
278            else:
279                cursor.movePosition(QTextCursor.MoveOperation.Start, move_mode)
280
281        # State transitions
282        elif self._vim_mode == VimMode.NORMAL:
283            if text == "i":
284                self._set_vim_mode(VimMode.INSERT)
285            elif text == "a":
286                cursor.movePosition(QTextCursor.MoveOperation.Right)
287                self._set_vim_mode(VimMode.INSERT)
288            elif text == "o":
289                cursor.movePosition(QTextCursor.MoveOperation.EndOfLine)
290                cursor.insertText("\n")
291                self._set_vim_mode(VimMode.INSERT)
292            elif text == "v":
293                self._set_vim_mode(VimMode.VISUAL)
294            elif text == "V":
295                self._set_vim_mode(VimMode.LINE_VISUAL)
296                cursor.movePosition(QTextCursor.MoveOperation.StartOfLine)
297                self._visual_anchor = cursor.position()
298                cursor.movePosition(
299                    QTextCursor.MoveOperation.EndOfLine, QTextCursor.MoveMode.KeepAnchor
300                )
301            elif text == ":":
302                self.vim_command_requested.emit()
303            elif text == "u":
304                self.undo()
305            elif text == "x":
306                cursor.deleteChar()
307            elif text == "p":
308                self.paste()
309            else:
310                # If no Vim command matches, allow standard processing
311                super().keyPressEvent(event)
312                return
313
314        elif self._vim_mode in (VimMode.VISUAL, VimMode.LINE_VISUAL):
315            if text == "y":
316                self.copy()
317                self._set_vim_mode(VimMode.NORMAL)
318            elif text in ("d", "x"):
319                self.cut()
320                self._set_vim_mode(VimMode.NORMAL)
321            elif text == "c":
322                self.cut()
323                self._set_vim_mode(VimMode.INSERT)
324            else:
325                super().keyPressEvent(event)
326                return
327
328        self.setTextCursor(cursor)

446    def _find_next(self, query: str, case: bool, whole: bool, regex: bool) -> bool:
447        """Locate the next occurrence of text matching the search criteria."""
448        flags = QTextDocument.FindFlag(0)
449        if case:
450            flags |= QTextDocument.FindFlag.FindCaseSensitively
451        if whole:
452            flags |= QTextDocument.FindFlag.FindWholeWords
453
454        if regex:
455            from PySide6.QtCore import QRegularExpression
456
457            re_flags = QRegularExpression.PatternOption.NoPatternOption
458            if not case:
459                re_flags |= QRegularExpression.PatternOption.CaseInsensitiveOption
460
461            rx = QRegularExpression(query, re_flags)
462            found = self.find(rx, flags)
463        else:
464            found = self.find(query, flags)
465
466        if not found:
467            # Wrap around from the start of the document
468            cursor = self.textCursor()
469            cursor.movePosition(QTextCursor.MoveOperation.Start)
470            self.setTextCursor(cursor)
471            if regex:
472                from PySide6.QtCore import QRegularExpression
473
474                re_flags = QRegularExpression.PatternOption.NoPatternOption
475                if not case:
476                    re_flags |= QRegularExpression.PatternOption.CaseInsensitiveOption
477                found = self.find(QRegularExpression(query, re_flags), flags)
478            else:
479                found = self.find(query, flags)
480        return found

482    def _replace(self, query: str, replace_text: str, case: bool, whole: bool, regex: bool) -> None:
483        """Replace the current selection if it matches, otherwise find next."""
484        cursor = self.textCursor()
485        if not cursor.hasSelection():
486            self._find_next(query, case, whole, regex)
487            return
488
489        selected = cursor.selectedText()
490        match = False
491        if regex:
492            import re
493
494            re_flags = 0 if case else re.IGNORECASE
495            if re.fullmatch(query, selected, flags=re_flags):
496                match = True
497        else:
498            if case:
499                match = selected == query
500            else:
501                match = selected.lower() == query.lower()
502
503        if match:
504            cursor.insertText(replace_text)
505            self._find_next(query, case, whole, regex)
506        else:
507            self._find_next(query, case, whole, regex)
508
509    def _replace_all(
510        self, query: str, replace_text: str, case: bool, whole: bool, regex: bool
511    ) -> None:
512        """Replace all occurrences in the entire document."""
513        old_cursor = self.textCursor()
514
515        cursor = self.textCursor()
516        cursor.movePosition(QTextCursor.MoveOperation.Start)
517        self.setTextCursor(cursor)
518
519        while self._find_next(query, case, whole, regex):
520            self.textCursor().insertText(replace_text)
521
522        self.setTextCursor(old_cursor)

447    def _find_next(self, query: str, case: bool, whole: bool, regex: bool) -> bool:
448        """Locate the next occurrence of text matching the search criteria."""
449        flags = QTextDocument.FindFlag(0)
450        if case:
451            flags |= QTextDocument.FindFlag.FindCaseSensitively
452        if whole:
453            flags |= QTextDocument.FindFlag.FindWholeWords
454
455        if regex:
456            from PySide6.QtCore import QRegularExpression
457
458            re_flags = QRegularExpression.PatternOption.NoPatternOption
459            if not case:
460                re_flags |= QRegularExpression.PatternOption.CaseInsensitiveOption
461
462            rx = QRegularExpression(query, re_flags)
463            found = self.find(rx, flags)
464        else:
465            found = self.find(query, flags)
466
467        if not found:
468            # Wrap around from the start of the document
469            cursor = self.textCursor()
470            cursor.movePosition(QTextCursor.MoveOperation.Start)
471            self.setTextCursor(cursor)
472            if regex:
473                from PySide6.QtCore import QRegularExpression
474
475                re_flags = QRegularExpression.PatternOption.NoPatternOption
476                if not case:
477                    re_flags |= QRegularExpression.PatternOption.CaseInsensitiveOption
478                found = self.find(QRegularExpression(query, re_flags), flags)
479            else:
480                found = self.find(query, flags)
481        return found

483    def _replace(self, query: str, replace_text: str, case: bool, whole: bool, regex: bool) -> None:
484        """Replace the current selection if it matches, otherwise find next."""
485        cursor = self.textCursor()
486        if not cursor.hasSelection():
487            self._find_next(query, case, whole, regex)
488            return
489
490        selected = cursor.selectedText()
491        match = False
492        if regex:
493            import re
494
495            re_flags = 0 if case else re.IGNORECASE
496            if re.fullmatch(query, selected, flags=re_flags):
497                match = True
498        else:
499            if case:
500                match = selected == query
501            else:
502                match = selected.lower() == query.lower()
503
504        if match:
505            cursor.insertText(replace_text)
506            self._find_next(query, case, whole, regex)
507        else:
508            self._find_next(query, case, whole, regex)

511        self, query: str, replace_text: str, case: bool, whole: bool, regex: bool
512    ) -> None:
513        """Replace all occurrences in the entire document."""
514        old_cursor = self.textCursor()
515

517        cursor.movePosition(QTextCursor.MoveOperation.Start)
518        self.setTextCursor(cursor)
