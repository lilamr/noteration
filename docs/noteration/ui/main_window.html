 412    def _save_session(self) -> None:
 413        if not self.config.get("general", "restore_last_session", True):
 414            return
 415
 416        tabs_data = []
 417        for pane, widget in self._iter_tabs():
 418            state = cast(NoterationTab, widget).session_state()
 419            if not state:
 420                continue
 421            state["pane"] = "split" if pane is self.tabs_split else "main"
 422            tabs_data.append(state)
 423
 424        active_pane = (
 425            "split"
 426            if self._active_tab_widget is self.tabs_split and self.tabs_split.count() > 0
 427            else "main"
 428        )
 429        self._session_store.save(tabs_data, active_pane)
 430
 431    def _restore_session(self) -> None:
 432        if not self.config.get("general", "restore_last_session", True):
 433            return
 434
 435        session = self._session_store.load()
 436        open_tabs = session.get("open_tabs", [])
 437        if not isinstance(open_tabs, list):
 438            return
 439
 440        for entry in open_tabs:
 441            if not isinstance(entry, dict):
 442                continue
 443
 444            pane_name = entry.get("pane", "main")
 445            target_pane = self.tabs_split if pane_name == "split" else self.tabs
 446            self._active_tab_widget = target_pane
 447
 448            tab_type = entry.get("type")
 449            if tab_type == "editor":
 450                path_value = entry.get("path")
 451                if not isinstance(path_value, str):
 452                    continue
 453                path = self.vault_path / path_value
 454                if path.exists():
 455                    if target_pane is self.tabs_split:
 456                        self.tabs_split.show()
 457                    self._open_note(path)
 458            elif tab_type == "pdf":
 459                path_value = entry.get("path")
 460                if not isinstance(path_value, str):
 461                    continue
 462                papis_key = entry.get("papis_key", "")
 463                if not isinstance(papis_key, str):
 464                    papis_key = ""
 465                path = self.vault_path / path_value
 466                if path.exists():
 467                    if target_pane is self.tabs_split:
 468                        self.tabs_split.show()
 469                    self._open_pdf(path, papis_key)
 470            elif tab_type == "literature":
 471                if target_pane is self.tabs_split:
 472                    self.tabs_split.show()
 473                self._open_literature_tab()
 474            elif tab_type == "sync":
 475                if target_pane is self.tabs_split:
 476                    self.tabs_split.show()
 477                self._open_sync_tab()
 478
 479        active_pane = session.get("active_pane", "main")
 480        if active_pane == "split" and self.tabs_split.count() > 0:
 481            self._active_tab_widget = self.tabs_split
 482            self.tabs_split.show()
 483        else:
 484            self._active_tab_widget = self.tabs
 485
 486        if self.tabs_split.count() == 0:
 487            self.tabs_split.hide()

 501    def _iter_tabs(self, reverse: bool = False):
 502        """Generator that yields (pane, widget) for every tab in all panes."""
 503        for pane in (self.tabs, self.tabs_split):
 504            indices: list[int] = list(range(pane.count()))
 505            if reverse:
 506                indices.reverse()
 507            for i in indices:
 508                yield pane, pane.widget(i)

 510    def _foreach_tab(self, callback: Callable[[QWidget], None]) -> None:
 511        """Execute a callback for every widget in all tab panes."""
 512        for _, w in self._iter_tabs():
 513            if w:
 514                callback(w)

 516    def _find_tab(
 517        self, tab_type: type[T], predicate: Optional[Callable[[T], bool]] = None, reverse: bool = False
 518    ) -> Optional[T]:
 519        """Find an existing tab of a specific type that matches an optional predicate.
 520
 521        If found, switches focus to that tab and returns it.
 522        """
 523        for pane, w in self._iter_tabs(reverse=reverse):
 524            if isinstance(w, tab_type):
 525                if predicate is None or predicate(w):
 526                    idx = pane.indexOf(w)
 527                    pane.setCurrentIndex(idx)
 528                    self._active_tab_widget = pane
 529                    return w
 530        return None

 544    def _open_note(self, path: Path) -> None:
 545        # Avoid opening the same note twice across both panes
 546        existing = self._find_tab(EditorTab, lambda w: w.file_path == path)
 547        if existing:
 548            # History tracking for existing tab
 549            if not self._is_navigating:
 550                current_w = self._active_tab_widget.currentWidget()
 551                if isinstance(current_w, EditorTab) and current_w.file_path != path:
 552                    self._history.append(current_w.file_path)
 553                    self._forward_stack.clear()
 554            return
 555
 556        # History tracking for new tab
 557        if not self._is_navigating:
 558            current_w = self._active_tab_widget.currentWidget()
 559            if isinstance(current_w, EditorTab):
 560                self._history.append(current_w.file_path)
 561                self._forward_stack.clear()
 562
 563        tab = EditorTab(path, self.vault)
 564        tab.cursor_moved.connect(self._on_cursor_moved)
 565        tab.content_changed.connect(lambda: self._mark_modified(tab))
 566        tab.wiki_link_clicked.connect(self._follow_wiki_link)
 567        tab.headings_changed.connect(self.sidebar.update_outline)
 568        tab.citations_changed.connect(self.sidebar.update_citations)
 569        tab.citations_changed.connect(self.sidebar.update_cited_pdfs)
 570        tab.word_count_changed.connect(self._on_word_count)

 413    def _save_session(self) -> None:
 414        if not self.config.get("general", "restore_last_session", True):
 415            return
 416
 417        tabs_data = []
 418        for pane, widget in self._iter_tabs():
 419            state = cast(NoterationTab, widget).session_state()
 420            if not state:
 421                continue
 422            state["pane"] = "split" if pane is self.tabs_split else "main"
 423            tabs_data.append(state)
 424
 425        active_pane = (
 426            "split"
 427            if self._active_tab_widget is self.tabs_split and self.tabs_split.count() > 0
 428            else "main"
 429        )
 430        self._session_store.save(tabs_data, active_pane)

 432    def _restore_session(self) -> None:
 433        if not self.config.get("general", "restore_last_session", True):
 434            return
 435
 436        session = self._session_store.load()
 437        open_tabs = session.get("open_tabs", [])
 438        if not isinstance(open_tabs, list):
 439            return
 440
 441        for entry in open_tabs:
 442            if not isinstance(entry, dict):
 443                continue
 444
 445            pane_name = entry.get("pane", "main")
 446            target_pane = self.tabs_split if pane_name == "split" else self.tabs
 447            self._active_tab_widget = target_pane
 448
 449            tab_type = entry.get("type")
 450            if tab_type == "editor":
 451                path_value = entry.get("path")
 452                if not isinstance(path_value, str):
 453                    continue
 454                path = self.vault_path / path_value
 455                if path.exists():
 456                    if target_pane is self.tabs_split:
 457                        self.tabs_split.show()
 458                    self._open_note(path)
 459            elif tab_type == "pdf":
 460                path_value = entry.get("path")
 461                if not isinstance(path_value, str):
 462                    continue
 463                papis_key = entry.get("papis_key", "")
 464                if not isinstance(papis_key, str):
 465                    papis_key = ""
 466                path = self.vault_path / path_value
 467                if path.exists():
 468                    if target_pane is self.tabs_split:
 469                        self.tabs_split.show()
 470                    self._open_pdf(path, papis_key)
 471            elif tab_type == "literature":
 472                if target_pane is self.tabs_split:
 473                    self.tabs_split.show()
 474                self._open_literature_tab()
 475            elif tab_type == "sync":
 476                if target_pane is self.tabs_split:
 477                    self.tabs_split.show()
 478                self._open_sync_tab()
 479
 480        active_pane = session.get("active_pane", "main")
 481        if active_pane == "split" and self.tabs_split.count() > 0:
 482            self._active_tab_widget = self.tabs_split
 483            self.tabs_split.show()
 484        else:
 485            self._active_tab_widget = self.tabs
 486
 487        if self.tabs_split.count() == 0:
 488            self.tabs_split.hide()

 502    def _iter_tabs(self, reverse: bool = False):
 503        """Generator that yields (pane, widget) for every tab in all panes."""
 504        for pane in (self.tabs, self.tabs_split):
 505            indices: list[int] = list(range(pane.count()))
 506            if reverse:
 507                indices.reverse()
 508            for i in indices:

 517    def _find_tab(
 518        self, tab_type: type[T], predicate: Optional[Callable[[T], bool]] = None, reverse: bool = False
 519    ) -> Optional[T]:
 520        """Find an existing tab of a specific type that matches an optional predicate.
 521
 522        If found, switches focus to that tab and returns it.
 523        """
 524        for pane, w in self._iter_tabs(reverse=reverse):
 525            if isinstance(w, tab_type):
 526                if predicate is None or predicate(w):
 527                    idx = pane.indexOf(w)
 528                    pane.setCurrentIndex(idx)
 529                    self._active_tab_widget = pane
 530                    return w
 531        return None

 651    def _on_extract_annotations(self, tab: PdfViewerTab) -> None:
 652        """Create a new Markdown note from all annotations in the PDF."""
 653        content = tab._doc_ann.compile_to_markdown(self.vault_path)
 654        if not content:
 655            QMessageBox.information(
 656                self, "No Annotations", "There are no annotations to extract from this PDF."
 657            )
 658            return
 659
 660        # Propose a filename
 661        base_name = f"Annotations_{tab.papis_key}"
 662        dest_path = self.vault_path / "notes" / f"{base_name}.md"
 663
 664        # Handle collision
 665        counter = 1
 666        while dest_path.exists():
 667            dest_path = self.vault_path / "notes" / f"{base_name}_{counter}.md"
 668            counter += 1
 669
 670        try:
 671            dest_path.write_text(content, encoding="utf-8")
 672            self.statusBar().showMessage(f"Extracted annotations to {dest_path.name}", 3000)
 673            self.sidebar.add_note(dest_path)
 674            self._open_note(dest_path)
 675        except Exception as e:
 676            logger.exception(f"Failed to extract annotations to {dest_path}: {e}")
 677            QMessageBox.critical(self, "Extraction Failed", f"Failed to save extracted note: {e}")

1416    def closeEvent(self, event) -> None:
1417        """Ensure all resources are cleaned up before closing."""
1418        logger.info("Main window closing...")
1419        self._save_session()
1420
1421        # 0. Disconnect global focus listener
1422        if self._focus_listener_connected:
1423            from PySide6.QtWidgets import QApplication
1424
1425            app = QApplication.instance()
1426            if isinstance(app, QApplication):
1427                try:
1428                    app.focusChanged.disconnect(self._on_focus_changed)
1429                except (TypeError, RuntimeError):
1430                    pass
1431            self._focus_listener_connected = False
1432
1433        # 1. Stop update thread if running
1434        if self._update_thread and self._update_thread.isRunning():
1435            self._update_thread.quit()
1436            if not self._update_thread.wait(1000):
1437                self._update_thread.terminate()
1438            self._update_thread = None
1439
1440        # 2. Stop tab threads and save dirty tabs
1441        for _, w in self._iter_tabs():
1442            if isinstance(w, NoterationTab):
1443                w.save_if_dirty()
1444                w.shutdown()
1445
1446        # 3. Shutdown vault manager (stops background threads + performs final save_all)
