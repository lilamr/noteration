246    def save(self) -> None:
247        if not self.is_modified:
248            return
249
250        # 1. Defensively check for existing save operation
251        if getattr(self, "_is_saving", False):
252            return
253
254        if self._save_thread:
255            try:
256                if self._save_thread.isRunning():
257                    return
258            except RuntimeError:
259                # C++ object was already deleted, nullify reference
260                self._save_thread = None
261                self._save_worker = None
262
263        text = self._editor.toPlainText()
264        # Reset modified flag immediately (optimistic UI)
265        self.is_modified = False
266        self._is_saving = True
267        self._update_focus_status()
268
269        # 2. Create background worker via manager
270        self._save_thread = QThread()
271        self._save_worker = self.vault.save_note(self.file_path, text)
272        self._save_worker.moveToThread(self._save_thread)
273
274        # 3. Connect signals
275        self._save_worker.finished.connect(self._on_save_success)
276        self._save_worker.error.connect(self._on_save_error)
277
278        # 4. Standard Cleanup Pattern
279        self._save_worker.finished.connect(self._save_thread.quit)
280        self._save_thread.finished.connect(self._clear_save_worker)
281
282        self._save_thread.started.connect(self._save_worker.run)
283        self._save_thread.start()
284
285        # Ensure the file is tracked by Git (status check is already async)
286        if self.vault.core.git_repo:
287            self.vault.request_git_status()

316    def shutdown(self) -> None:
317        """Wait for background saving to finish if active and cleanup resources."""
318        # 1. Stop background saving
319        if self._save_thread:
320            try:
321                if self._save_thread.isRunning():
322                    logger.info(f"EditorTab: Waiting for save to finish for {self.file_path}...")
323                    if self._save_thread.wait(5000):
324                        self._clear_save_worker()
325                    else:
326                        logger.error(f"EditorTab: Save thread for {self.file_path} timed out. Holding reference.")
327                else:
328                    self._clear_save_worker()
329            except RuntimeError:
330                # Already deleted, just clear references
331                self._save_thread = None
332                self._save_worker = None
333        else:
334            self._clear_save_worker()
335
336        # 2. Cleanup other resources
337        if hasattr(self, "_debounce") and self._debounce:
338            self._debounce.stop()
339        if hasattr(self, "_preview") and self._preview:
340            self._preview.shutdown()

560    def go_to_heading(self, heading_text: str) -> None:
561        text = self._editor.toPlainText()
562        pattern = f"^#+\\s*{re.escape(heading_text)}$"
563        for match in re.finditer(pattern, text, re.MULTILINE):
564            cursor = self._editor.textCursor()
565            cursor.setPosition(match.start())
566            self._editor.setTextCursor(cursor)
567            self._editor.setFocus()
568            return

570    def go_to_citation(self, key: str) -> None:
571        text = self._editor.toPlainText()
572        # Match @key optionally followed by [locator]
573        pattern = f"@{re.escape(key)}(?:\\[[^\\]]+\\])?\\b"
574        match = re.search(pattern, text, re.IGNORECASE)
575        if match:
576            cursor = self._editor.textCursor()
577            cursor.setPosition(match.start())
578            self._editor.setTextCursor(cursor)

589    def word_count(self) -> int:
590        """Calculate word count, excluding YAML front-matter and code blocks."""
591        text = self._editor.toPlainText()
592        text = re.sub(r"^---\n.*?\n---\n", "", text, flags=re.DOTALL)
593        text = re.sub(r"```.*?```", "", text, flags=re.DOTALL)
594        text = re.sub(r"`[^`]+`", "", text)
595        return len(re.findall(r"\b\w+\b", text))
