 76class HeadingModel(QAbstractItemModel):
 77    """Hierarchical model for the note Outline (Headings)."""
 78
 79    def __init__(self, parent=None) -> None:
 80        super().__init__(parent)
 81        self.root_item = HeadingItem(0, "Root")
 82
 83    def set_headings(self, headings: list[tuple[int, str]]) -> None:
 84        """Update the hierarchy based on a list of (level, title) tuples."""
 85        self.beginResetModel()
 86        self.root_item = HeadingItem(0, "Root")
 87        stack = [self.root_item]
 88
 89        for level, text in headings:
 90            while len(stack) > 1 and stack[-1].level >= level:
 91                stack.pop()
 92
 93            new_item = HeadingItem(level, text, stack[-1])
 94            stack[-1].append_child(new_item)
 95            stack.append(new_item)
 96
 97        self.endResetModel()
 98
 99    def index(
100        self, row: int, column: int, parent: QModelIndex | QPersistentModelIndex = QModelIndex()
101    ) -> QModelIndex:
102        if not self.hasIndex(row, column, parent):
103            return QModelIndex()
104
105        parent_item = parent.internalPointer() if parent.isValid() else self.root_item
106        child_item = parent_item.child(row)
107        if child_item:
108            return self.createIndex(row, column, child_item)
109        return QModelIndex()
110
111    def parent(self, index: QModelIndex | QPersistentModelIndex) -> QModelIndex:  # type: ignore[override]
112        if not index.isValid():
113            return QModelIndex()
114
115        child_item = index.internalPointer()
116        parent_item = child_item.parent_item
117
118        if parent_item == self.root_item or parent_item is None:
119            return QModelIndex()
120
121        return self.createIndex(parent_item.row(), 0, parent_item)
122
123    def rowCount(self, parent: QModelIndex | QPersistentModelIndex = QModelIndex()) -> int:
124        if parent.column() > 0:
125            return 0
126        parent_item = parent.internalPointer() if parent.isValid() else self.root_item
127        return parent_item.child_count()
128
129    def columnCount(self, parent: QModelIndex | QPersistentModelIndex = QModelIndex()) -> int:
130        return 1
131
132    def data(
133        self, index: QModelIndex | QPersistentModelIndex, role: int = Qt.ItemDataRole.DisplayRole
134    ) -> Any:
135        if not index.isValid():
136            return None
137        item = index.internalPointer()
138        if role == Qt.ItemDataRole.DisplayRole:
139            prefix = "#" * item.level + " "
140            return prefix + item.text
141        if role == Qt.ItemDataRole.UserRole:
142            return item.text
143        return None

216    def populate(self) -> None:
217        """Populate the tree based on the notes/ folder structure with persistent ordering."""
218        self.clear()
219        if not self.root_path.exists():
220            return
221
222        order_data = self.load_order()
223        if order_data:
224            self._apply_order(order_data, self.root_path, self.invisibleRootItem())
225            # Check for new items that were manually added and are not in order_data
226            self._scan_remaining(self.root_path, self.invisibleRootItem())
227        else:
228            self._scan_dir(self.root_path, self.invisibleRootItem())

230    def _scan_dir(self, path: Path, parent_item: QTreeWidgetItem) -> None:
231        """Initial alphabetical scan of the directory structure."""
232        entries = sorted(list(path.iterdir()), key=lambda x: (not x.is_dir(), x.name.lower()))
233
234        for entry in entries:
235            if entry.is_dir():
236                item = QTreeWidgetItem(parent_item, [entry.name])
237                item.setData(0, Qt.ItemDataRole.UserRole, entry)
238                item.setIcon(0, self.style().standardIcon(self.style().StandardPixmap.SP_DirIcon))
239                self._scan_dir(entry, item)
240            elif entry.suffix.lower() == ".md":
241                item = QTreeWidgetItem(parent_item, [entry.name])
242                item.setData(0, Qt.ItemDataRole.UserRole, entry)
243                item.setIcon(0, self.style().standardIcon(self.style().StandardPixmap.SP_FileIcon))
244
245    def _apply_order(
246        self, order_list: list[dict], current_path: Path, parent_item: QTreeWidgetItem
247    ) -> None:
248        """Recursively apply the stored JSON ordering to the tree."""
249        for entry_info in order_list:
250            name = entry_info.get("name")
251            if not name:
252                continue
253            is_dir = entry_info.get("is_dir", False)
254            path = current_path / str(name)
255
256            if not path.exists():
257                continue
258
259            item = QTreeWidgetItem(parent_item, [str(name)])
260            item.setData(0, Qt.ItemDataRole.UserRole, path)
261
262            if is_dir:
263                item.setIcon(0, self.style().standardIcon(self.style().StandardPixmap.SP_DirIcon))
264                children = entry_info.get("children", [])
265                self._apply_order(children, path, item)
266            else:
267                item.setIcon(0, self.style().standardIcon(self.style().StandardPixmap.SP_FileIcon))
268
269    def _scan_remaining(self, path: Path, parent_item: QTreeWidgetItem) -> None:
270        """Scan for items on disk not yet present in the UI."""
271        existing_names = set()
272        for i in range(parent_item.childCount()):
273            existing_names.add(parent_item.child(i).text(0))
274
275        for entry in path.iterdir():
276            if entry.name not in existing_names:
277                if entry.is_dir():
278                    item = QTreeWidgetItem(parent_item, [entry.name])
279                    item.setData(0, Qt.ItemDataRole.UserRole, entry)
280                    item.setIcon(
281                        0, self.style().standardIcon(self.style().StandardPixmap.SP_DirIcon)
282                    )
283                    self._scan_dir(entry, item)
284                elif entry.suffix.lower() == ".md":
285                    item = QTreeWidgetItem(parent_item, [entry.name])
286                    item.setData(0, Qt.ItemDataRole.UserRole, entry)
287                    item.setIcon(
288                        0, self.style().standardIcon(self.style().StandardPixmap.SP_FileIcon)
289                    )
290            elif entry.is_dir():
291                # Recursively check the contents of existing folders
292                child_item = self.find_item_by_path(entry, parent_item)
293                if child_item:
294                    self._scan_remaining(entry, child_item)
295
296    def save_order(self) -> None:
297        """Persist the current tree structure to a JSON file."""
298        data = self._get_item_data(self.invisibleRootItem())
299        try:
300            self._order_file.parent.mkdir(parents=True, exist_ok=True)
301            with open(self._order_file, "w", encoding="utf-8") as f:
302                json.dump(data, f, indent=2)
303        except Exception as e:
304            print(f"Error saving order: {e}")
305
306    def load_order(self) -> list[dict] | None:
307        """Load the persisted tree structure from JSON."""
308        if not self._order_file.exists():
309            return None
310        try:
311            with open(self._order_file, "r", encoding="utf-8") as f:
312                return json.load(f)
313        except Exception as e:
314            print(f"Error loading order: {e}")
315            return None
316
317    def _get_item_data(self, parent_item: QTreeWidgetItem) -> list[dict]:
318        """Convert tree items to a serializable dictionary structure."""
319        items = []
320        for i in range(parent_item.childCount()):
321            child = parent_item.child(i)
322            path = child.data(0, Qt.ItemDataRole.UserRole)
323            item_info = {
324                "name": child.text(0),
325                "is_dir": path.is_dir(),
326            }
327            if item_info["is_dir"]:
328                item_info["children"] = self._get_item_data(child)
329            items.append(item_info)
330        return items
331
332    def _on_double_clicked(self, item: QTreeWidgetItem, column: int) -> None:
333        path = item.data(0, Qt.ItemDataRole.UserRole)
334        if isinstance(path, Path) and path.is_file():
335            self.note_selected.emit(path)
336
337    def find_item_by_path(
338        self, path: Path, parent: QTreeWidgetItem | None = None
339    ) -> QTreeWidgetItem | None:
340        """Recursively locate a tree item corresponding to a file path."""
341        root = parent or self.invisibleRootItem()
342        if root.data(0, Qt.ItemDataRole.UserRole) == path:
343            return root
344
345        for i in range(root.childCount()):
346            child = root.child(i)
347            res = self.find_item_by_path(path, child)
348            if res:
349                return res
350        return None

352    def add_item_to_ui(self, path: Path) -> None:
353        """Add new item to the bottom position of its parent in UI if not already present."""
354        if self.find_item_by_path(path):
355            return
356
357        parent_path = path.parent
358        parent_item = self.find_item_by_path(parent_path) or self.invisibleRootItem()
359
360        item = QTreeWidgetItem(parent_item, [path.name])
361        item.setData(0, Qt.ItemDataRole.UserRole, path)
362
363        if path.is_dir():
364            item.setIcon(0, self.style().standardIcon(self.style().StandardPixmap.SP_DirIcon))
365        else:
366            item.setIcon(0, self.style().standardIcon(self.style().StandardPixmap.SP_FileIcon))
367
368        self.setCurrentItem(item)
369        self.scrollToItem(item)

372    def dropEvent(self, event) -> None:
373        """Handle file/folder relocation on disk after a drag-and-drop operation."""
374        selected_item = self.currentItem()
375        if not selected_item:
376            super().dropEvent(event)
377            return
378
379        old_path = selected_item.data(0, Qt.ItemDataRole.UserRole)
380
381        # Execute the default drop behavior to update the UI
382        super().dropEvent(event)
383
384        # Calculate the new path based on the drop target
385        new_parent_item = selected_item.parent() or self.invisibleRootItem()
386        new_parent_path = new_parent_item.data(0, Qt.ItemDataRole.UserRole) or self.root_path
387
388        new_path = new_parent_path / old_path.name
389
390        if old_path != new_path:
391            try:
392                if old_path.exists():
393                    shutil.move(str(old_path), str(new_path))
394                    selected_item.setData(0, Qt.ItemDataRole.UserRole, new_path)
395                    self.item_moved.emit(old_path, new_path)
396                    # Recursively update paths if a directory was moved
397                    if new_path.is_dir():
398                        self._update_item_paths(selected_item, new_path)
399            except Exception as e:
400                QMessageBox.critical(self, "Move Failed", str(e))
401                self.populate()  # Refresh to sync disk state
402
403        self.save_order()
404
405    def _update_item_paths(self, parent_item: QTreeWidgetItem, new_parent_path: Path) -> None:
406        """Update UserRole paths for all descendants of a moved directory."""
407        for i in range(parent_item.childCount()):
408            child = parent_item.child(i)
409            old_path = child.data(0, Qt.ItemDataRole.UserRole)
410            new_path = new_parent_path / old_path.name
411            child.setData(0, Qt.ItemDataRole.UserRole, new_path)
412            if child.childCount() > 0:
413                self._update_item_paths(child, new_path)
414
415    def get_selected_path(self) -> Path:
416        item = self.currentItem()
417        if not item:
418            return self.root_path
419        return item.data(0, Qt.ItemDataRole.UserRole)

484            if not is_folder and not new_name.endswith(".md"):
485                new_name += ".md"
486
487            new_path = old_path.parent / new_name
488            try:
489                old_path.rename(new_path)
490                item.setText(0, new_name)
491                item.setData(0, Qt.ItemDataRole.UserRole, new_path)
492                self.item_moved.emit(old_path, new_path)
493                self.tree.save_order()
494            except Exception as e:

497    def _delete_item(self, item: QTreeWidgetItem) -> None:
498        path = item.data(0, Qt.ItemDataRole.UserRole)
499        if not path.exists():
500            return
501
502        if (
503            QMessageBox.question(
504                self,
505                "Delete",
506                f"Permanently delete '{path.name}'?",
507                QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No,

509            == QMessageBox.StandardButton.Yes

511            try:
512                if path.is_dir():

514                else:
515                    path.unlink()
516                # Remove from the tree UI

519                self.tree.save_order()
520            except Exception as e:

588        """Scan the Papis library and build the tree UI."""
589        self.tree.clear()
590        lit_dir = self.config.papis_library
591        if not lit_dir.exists():
592            return
593
594        yaml = get_yaml()
595
596        all_collections = set()
597
598        # 1. Direct PDFs in library root
599        for pdf_file in sorted(lit_dir.glob("*.pdf")):
600            item = QTreeWidgetItem(self.tree, [f"📘 {pdf_file.name}"])
601            item.setData(
602                0,
603                Qt.ItemDataRole.UserRole,
604                {"key": pdf_file.stem, "pdf": pdf_file, "collections": []},
605            )
606
607        # 2. Folder-based entries with info.yaml
608        for entry_dir in sorted(lit_dir.iterdir()):
609            if not entry_dir.is_dir() or not (entry_dir / "info.yaml").exists():
610                continue
611            try:
612                if yaml is None:
613                    continue
614                with open(entry_dir / "info.yaml") as f:
615                    info = yaml.safe_load(f)
616                title = info.get("title", entry_dir.name)[:40]
617                colls = info.get("collections", [])
618                for c in colls:
619                    all_collections.add(str(c))
620                pdf = list(entry_dir.glob("*.pdf"))
621                item = QTreeWidgetItem(self.tree, [f"{'📘' if pdf else '📂'} {title}"])
622                item.setData(
623                    0,
624                    Qt.ItemDataRole.UserRole,
625                    {
626                        "key": entry_dir.name,
627                        "pdf": pdf[0] if pdf else None,
628                        "collections": [str(c) for c in colls],
629                    },
630                )
631            except Exception as e:
632                logger.warning(f"Failed to process Papis entry directory {entry_dir}: {e}")
633
634        # Update collection filter dropdown
635        curr = self.combo.currentText()
636        self.combo.blockSignals(True)
637        self.combo.clear()
638        self.combo.addItems(["All", "Linked"] + sorted(list(all_collections)))
639        if curr in ["All", "Linked"] or curr in all_collections:
640            self.combo.setCurrentText(curr)
641        self.combo.blockSignals(False)
642        self._filter()
