 55class LiteratureFilterProxy(QSortFilterProxyModel):
 56    """Proxy model for advanced filtering and sorting of literature."""
 57
 58    def __init__(self, parent=None):
 59        super().__init__(parent)
 60        self.setFilterCaseSensitivity(Qt.CaseSensitivity.CaseInsensitive)
 61        self._collection_filter = "All"

 63    def set_collection_filter(self, collection: str):
 64        self._collection_filter = collection
 65        self.invalidateFilter()
 66
 67    def filterAcceptsRow(self, source_row, source_parent):
 68        model = self.sourceModel()
 69        entry = model.get_entry(source_row)
 70        if not entry:
 71            return False
 72
 73        # 1. Collection Filter
 74        if self._collection_filter != "All":
 75            if self._collection_filter not in entry.collections:
 76                return False
 77
 78        # 2. Text Search Filter (multi-token AND)
 79        query = self.filterRegularExpression().pattern().lower()
 80        if not query:
 81            return True
 82
 83        tokens = query.split()
 84        entry_text = (
 85            f"{entry.key} {entry.title} {entry.author} {entry.year} {' '.join(entry.tags)}".lower()
 86        )
 87
 88        return all(token in entry_text for token in tokens)

 91class LiteratureModel(QAbstractListModel):
 92    """Efficient model for literature entries to handle large libraries."""
 93
 94    def __init__(self, entries: list[LiteratureEntry] | None = None, parent=None):
 95        super().__init__(parent)
 96        self._entries = entries or []
 97
 98    def rowCount(self, parent=QModelIndex()):
 99        return len(self._entries)
100
101    def data(self, index, role=Qt.ItemDataRole.DisplayRole):
102        if not index.isValid() or not (0 <= index.row() < len(self._entries)):
103            return None
104
105        entry = self._entries[index.row()]
106
107        if role == Qt.ItemDataRole.DisplayRole:
108            coll_str = f" [{', '.join(entry.collections)}]" if entry.collections else ""
109            return f"@{entry.key}{coll_str}\n{entry.title[:55]}\n{entry.author[:35]} · {entry.year}"
110
111        elif role == Qt.ItemDataRole.UserRole:
112            return entry
113
114        return None
115
116    def set_entries(self, entries: list[LiteratureEntry]):
117        self.beginResetModel()
118        self._entries = entries
119        self.endResetModel()
120
121    def get_entry(self, row: int) -> LiteratureEntry | None:
122        if 0 <= row < len(self._entries):
123            return self._entries[row]
124        return None
